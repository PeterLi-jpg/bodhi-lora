\usepackage{tikz}

% ── Palette (Tol bright, matches every other figure in the paper) ────────────
\definecolor{colBlue}{HTML}{0077BB}
\definecolor{colTeal}{HTML}{009988}
\definecolor{colOrange}{HTML}{EE7733}
\definecolor{colRed}{HTML}{CC3311}
\definecolor{colGray}{HTML}{444444}

\tikzset{
  % Training-pipeline step boxes
  pbox/.style={
    draw, very thick, rounded corners=4pt, align=center,
    inner sep=8pt, minimum width=3.0cm, minimum height=1.45cm, font=\small,
  },
  pA/.style={pbox, draw=colBlue!80!black,   fill=colBlue!10},
  pB/.style={pbox, draw=colTeal!80!black,   fill=colTeal!10},
  pC/.style={pbox, draw=colOrange!80!black, fill=colOrange!12},
  pD/.style={pbox, draw=colOrange!80!black, fill=colOrange!12},
  % 2x2 cells
  cell/.style={
    draw, very thick, rounded corners=4pt, align=center,
    inner sep=7pt, minimum width=3.4cm, minimum height=1.05cm, font=\small,
  },
  cBase/.style={cell, draw=colRed!80!black,  fill=colRed!8},
  cLora/.style={cell, draw=colBlue!80!black, fill=colBlue!8},
  % Matrix row / column headers (no frame)
  rowhead/.style={align=right,  font=\small\bfseries, text=colGray},
  colhead/.style={align=center, font=\small\itshape, text=colGray},
  % Held-out evaluation set
  holdout/.style={
    draw, very thick, rounded corners=4pt, align=center, inner sep=8pt,
    minimum width=8.2cm, minimum height=1.1cm, font=\small,
    draw=colGray, fill=colGray!6,
  },
  arr/.style={-{Latex[length=2.8mm,width=2mm]}, very thick, colGray},
  arrmid/.style={-{Latex[length=2.5mm,width=1.8mm]}, very thick},
  region/.style={draw, dashed, thick, rounded corners=6pt, inner sep=11pt},
}

\begin{document}
\begin{tikzpicture}

%% ── Training pipeline: one straight left-to-right chain ────────────────────
\node[pA]                 (s1) {%
  \textbf{1 · Generate}\\[1pt] BODHI CoT on\\ MedGemma-27B\\
  {\footnotesize 4,000 prompts, greedy}};
\node[pB, right=9mm of s1] (s2) {%
  \textbf{2 · Grade}\\[1pt] Qwen-14B rubric\\ per-criterion\\
  {\footnotesize score $s\!\in\![0,1]$}};
\node[pC, right=9mm of s2] (s3) {%
  \textbf{3 · Filter}\\[1pt] Retain $s \geq 0.4$\\ 90/10 train--val\\
  {\footnotesize $\sim$3,100 traces/seed}};
\node[pD, right=9mm of s3] (s4) {%
  \textbf{4 · Distill}\\[1pt] LoRA SFT\\ $r{=}16$, 3 epochs\\
  {\footnotesize 5 seeds}};

\draw[arr] (s1) -- (s2);
\draw[arr] (s2) -- (s3);
\draw[arr] (s3) -- (s4);

%% ── Evaluation: the 2x2 factorial drawn as an actual 2x2 ──────────────────
%% A matrix aligns the cells exactly, so no hand-computed offsets are needed.
\matrix (m) [
    below=22mm of s2.south east,
    matrix, row sep=5mm, column sep=6mm,
    anchor=north,
  ]{
    \node[rowhead] {};                                        &
    \node[colhead] {no CoT at inference};                     &
    \node[colhead] {$+$ BODHI CoT at inference};              \\
    \node[rowhead] {Base model\\MedGemma-27B};                &
    \node[cBase] (ebase) {\textbf{Base}};                     &
    \node[cBase] (ebcot) {\textbf{Base $+$ BODHI CoT}};       \\
    \node[rowhead] (lorarow) {LoRA adapter\\(5 seeds)};        &
    \node[cLora] (elora) {\textbf{LoRA}};                     &
    \node[cLora] (elcot) {\textbf{LoRA $+$ BODHI CoT}};       \\
  };

%% The distilled adapter *is* the LoRA row, so the arrow targets that row label,
%% not one cell. Routed around the outside left so it crosses no text: the
%% horizontal leg sits in the empty band above the matrix, the vertical leg
%% outside the row labels.
\coordinate (a) at ($(s4.south)+(0,-11mm)$);            % below step 4
\coordinate (b) at ($(m.west)+(-13mm,0)$);              % clear of the row labels
\draw[arrmid, colOrange!80!black]
  (s4.south) -- (a) -- (a -| b) -- (b |- lorarow) -- (lorarow.west);

%% ── Held-out evaluation set: one arrow, not four ──────────────────────────
%% Centre the holdout under the four cells. m.south is left of centre because
%% the matrix includes the row-label column, which is what skewed the arrow.
\coordinate (cellsmid) at ($(elora)!0.5!(elcot)$);
\node[holdout] (hb) at ($(cellsmid |- m.south)+(0,-15mm)$) {%
  \textbf{HealthBench-Hard} (held-out)\\
  {\footnotesize 1,000 prompts · Llama-3.1-8B-Instruct grader}};

\draw[arr] ($(cellsmid |- m.south)+(0,-1mm)$) -- (hb.north);

%% ── Background regions ────────────────────────────────────────────────────
\begin{scope}[on background layer]
  \node[region, draw=colTeal!60!black, fill=colTeal!4,
        fit=(s1)(s4),
        label={[font=\footnotesize\bfseries, text=colTeal!70!black]
               above:{\strut Training pipeline}}] {};
  \node[region, draw=colRed!50!black, fill=colRed!3,
        fit=(m)(hb),
        label={[font=\footnotesize\bfseries, text=colRed!70!black]
               below:{\strut Evaluation ($2{\times}2$ factorial, out-of-sample)}}] {};
\end{scope}

\end{tikzpicture}
